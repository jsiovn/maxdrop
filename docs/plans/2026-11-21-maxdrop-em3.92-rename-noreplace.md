# Rename that never replaces in archive._rename_new Implementation Plan

> **For agentic workers:** Use subagents when appropriate to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Close the look-then-rename window of `_rename_new` (maxdrop/intake/archive.py) on POSIX: a folder another
process makes at the target name in that instant is never replaced.

**Architecture:** `_rename_new` first asks a platform rename that refuses an existing target (`_noreplace()`):
Windows `os.rename` (it refuses by itself), Linux `renameat2(AT_FDCWD, src, AT_FDCWD, dst, RENAME_NOREPLACE)` and
macOS `renamex_np(src, dst, RENAME_EXCL)` through ctypes, loaded once and cached. The primitive returns True when
renamed, False when the system or the file system does not support it (ENOSYS: old kernel; EINVAL / ENOTSUP /
EOPNOTSUPP: a file system without the flag), and raises OSError otherwise. False, or no primitive (another POSIX
system, a libc without the symbol), falls back to today's look-then-rename. The error contract of `_rename_new`
stays: False when dst exists, OSError raised otherwise (so `_create` falls back to `_make`). `os.link` + `os.unlink`
does not apply: src and dst are folders.

**Tech Stack:** Python 3.11 stdlib (ctypes, errno), unittest.

```python
_AT_FDCWD = -100           # Linux: paths relative to the working directory, as os.rename
_RENAME_NOREPLACE = 1      # Linux renameat2 flag
_RENAME_EXCL = 0x4         # macOS renamex_np flag
_NOREPLACE_UNSUPPORTED = frozenset(
    e for e in (errno.ENOSYS, errno.EINVAL, getattr(errno, "ENOTSUP", None), errno.EOPNOTSUPP) if e is not None)
_NOREPLACE = []            # cache: [the rename or None]

def _noreplace():
    if not _NOREPLACE:
        _NOREPLACE.append(_load_noreplace())
    return _NOREPLACE[0]

def _load_noreplace():
    if os.name == "nt":
        def rename(src, dst):
            os.rename(src, dst)
            return True
        return rename
    try:
        import ctypes
        libc = ctypes.CDLL(None, use_errno=True)
        if sys.platform.startswith("linux"):
            fn = libc.renameat2
            fn.argtypes = (ctypes.c_int, ctypes.c_char_p, ctypes.c_int, ctypes.c_char_p, ctypes.c_uint)
            def call(s, d): return fn(_AT_FDCWD, s, _AT_FDCWD, d, _RENAME_NOREPLACE)
        elif sys.platform == "darwin":
            fn = libc.renamex_np
            fn.argtypes = (ctypes.c_char_p, ctypes.c_char_p, ctypes.c_uint)
            def call(s, d): return fn(s, d, _RENAME_EXCL)
        else:
            return None
        fn.restype = ctypes.c_int
    except (ImportError, OSError, AttributeError):
        return None

    def rename(src, dst):
        if call(os.fsencode(src), os.fsencode(dst)) == 0:
            return True
        err = ctypes.get_errno()
        if err in _NOREPLACE_UNSUPPORTED:
            return False
        raise OSError(err, os.strerror(err), src, None, dst)
    return rename

def _rename_new(src, dst):
    rename = _noreplace()
    try:
        if rename is not None and rename(src, dst):
            return True
        if os.path.lexists(dst):
            return False
        os.rename(src, dst)
    except OSError:
        if os.path.lexists(dst):
            return False
        raise
    return True
```
