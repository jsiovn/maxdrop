```python
"""Recount the ParamBlock2 figures of docs/FORMAT.md (Parameter blocks) on the committed fixtures."""
import collections, glob, os, struct, sys, unittest
ROOT = sys.argv[1]
sys.path.insert(0, os.path.join(ROOT, "tests", "unit"))
import test_materials as tm  # noqa: E402  (builds the maxdrop_convert_test package)
cfb, scene, materials, classids = tm.cfb, tm.scene, tm.materials, tm.classids
HEAD = struct.Struct("<IHIHI")

def header(chunk):
    h, skip = chunk.first(0x0009), 0
    if h is None:
        h, skip = chunk.first(0x000B), 8
    return (0x0009 if skip == 0 else 0x000B,) + HEAD.unpack(h.view()[skip:skip + HEAD.size])

graph_v, scene_v = collections.Counter(), collections.Counter()
graph_u, scene_u, graph_tag, scene_tag = [], [], collections.Counter(), collections.Counter()
owner_ok = owner_bad = 0
graph_blocks = scene_blocks = 0
per_fixture = {}
for path in tm.FIXTURES:
    name = os.path.basename(path)[:-4]
    with cfb.CompoundFile(path) as cf:
        s = scene.Scene.from_file(cf)
        g = materials.MaterialGraph(s)
        order = g.walk(s.node(i).material for i in s.walk())
        in_graph = set()
        for i in order:
            m = g.get(i)
            holders = [(m.index, m.refs)] + [(p.index, s.refs(p.index)) for p in m.parts.values()]
            for owner, refs in holders:
                for slot, t in refs.items():
                    e = s.entry(t)
                    if e is not None and e.superclass == classids.SC_PARAMETER_BLOCK2:
                        in_graph.add((t, owner))
        for i in range(len(s)):
            e = s.entry(i)
            if e is None or e.superclass != classids.SC_PARAMETER_BLOCK2:
                continue
            tag, u, bid, ver, cnt, own = header(s.chunk(i))
            scene_blocks += 1; scene_v[ver] += 1; scene_u.append(u); scene_tag[tag] += 1
        nb = 0
        for t in sorted({t for t, _o in in_graph}):
            tag, u, bid, ver, cnt, own = header(s.chunk(t))
            owners = {o for tt, o in in_graph if tt == t}
            if own in owners: owner_ok += 1
            else: owner_bad += 1; print("owner mismatch", name, t, own, owners)
            graph_blocks += 1; nb += 1; graph_v[ver] += 1; graph_u.append(u); graph_tag[tag] += 1
        per_fixture[name] = nb
        del cf

print("graph blocks", graph_blocks, "versions", {hex(k): v for k, v in graph_v.most_common()},
      "u32 range", min(graph_u), max(graph_u), "headers", dict(graph_tag), "owner ok/bad", owner_ok, owner_bad)
print("scene blocks", scene_blocks, "versions", {hex(k): v for k, v in scene_v.most_common()},
      "u32 range", min(scene_u), max(scene_u), "headers", dict(scene_tag))

compared = {}
orig = tm.Parity.__init__
def init(self, *a, **k):
    orig(self, *a, **k)
    compared[a[0]._subtest_name] = self
tm.Parity.__init__ = init
class T(tm.Fixtures):
    def subTest(self, msg=None, **kw):
        self._subtest_name = msg
        return super().subTest(msg, **kw)
r = unittest.TextTestRunner(verbosity=0).run(unittest.defaultTestLoader.loadTestsFromName("test_raw_param_parity", T))
for n in sorted(compared):
    p = compared[n]
    print("%-22s compared %5d floor %5d animated %d blocks %d" % (n, p.compared, tm.FLOORS[n], p.animated, per_fixture[n]))
print("total compared", sum(p.compared for p in compared.values()))
```

`recount4.py`: the header rule (the test `HeaderOnFixtures` now pins it).

```python
"""0x0009 first u32 vs the owner's class-directory index (and not its object index); 0x000B 8 bytes vs owner ClassID."""
import collections, os, struct, sys
ROOT = sys.argv[1]
sys.path.insert(0, os.path.join(ROOT, "tests", "unit"))
import test_materials as tm
cfb, scene, classids = tm.cfb, tm.scene, tm.classids
HEAD = struct.Struct("<IHIHI")
c9 = collections.Counter(); c11 = collections.Counter()
for path in tm.FIXTURES:
    with cfb.CompoundFile(path) as cf:
        s = scene.Scene.from_file(cf)
        for i in range(len(s)):
            e = s.entry(i)
            if e is None or e.superclass != classids.SC_PARAMETER_BLOCK2: continue
            c = s.chunk(i); h = c.first(0x0009)
            if h is not None:
                u, bid, ver, cnt, own = HEAD.unpack(h.view()[:16]); oe = s.entry(own)
                c9[("u==owner class index", u == oe.index)] += 1
                c9[("u==owner chunk id", u == s.chunk(own).id)] += 1
                c9[("u==owner object index", u == own)] += 1
                c9[("block class index != owner's", e.index != oe.index)] += 1
            else:
                v = c.first(0x000B).view(); ca, cb = struct.unpack("<II", v[:8])
                u, bid, ver, cnt, own = HEAD.unpack(v[8:24]); oe = s.entry(own)
                c11[("8 bytes==owner ClassID", (ca, cb) == tuple(oe.class_id))] += 1
                c11[("u==owner superclass", u == oe.superclass)] += 1
        del cf
print(sorted(c9.items())); print(sorted(c11.items()))
```
